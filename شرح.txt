OPEN_TAG : '<' -> pushMode(HTML_MODE);
DOUBLE_BRACE : '{{' -> pushMode(JINJA_EXPR_MODE);
OPEN_BLOCK : '{%' -> pushMode(JINJA_BLOCK_MODE);
DEF: 'def';
FOR: 'for';
IF: 'if';

الوراثة (Inheritance)

public abstract class ASTNode {
    private final int line;      // رقم السطر

}

استخدام نمط الزائر (Visitor Pattern):

public interface ASTVisitor<T> {
    T visitDocumentNode(DocumentNode node);
    T visitHtmlElementNode(HtmlElementNode node);
    T visitFunctionDefNode(FunctionDefNode node);
    // ... المزيد
}
كل عقدة تنفذ

@Override
public HtmlElementNode visitPairedElement(HtmlParser.PairedElementContext ctx) {
    List<ElementNode> children = new ArrayList<>();
    if (ctx.element() != null) {
        ElementNode child = (ElementNode) visit(ctx.element());
        if (child != null) {
            children.add(child);
        }
    }
    return buildHtmlElement(ctx, ctx.htmlParts(), false, false, children);
}
مثال - زيارة تعريف دالة:

@Override
public FunctionDefNode visitFunc_def(HtmlParser.Func_defContext ctx) {
    Token token = ctx.getStart();
    String name = ctx.NAME().getText();
    
    // جمع المزخرفات
    List<DecoratorNode> decorators = new ArrayList<>();
    if (ctx.decorators() != null) {
        for (HtmlParser.DecoratorContext decoratorCtx : ctx.decorators().decorator()) {
            decorators.add((DecoratorNode) visit(decoratorCtx));
        }
    }
    
    // جمع المعاملات
    List<ParameterNode> parameters = new ArrayList<>();
    if (ctx.params() != null && ctx.params().param() != null) {
        for (HtmlParser.ParamContext paramCtx : ctx.params().param()) {
            parameters.add((ParameterNode) visit(paramCtx));
        }
    }
    
    // جمع جسم الدالة
    List<StatementNode> body = new ArrayList<>();
    if (ctx.suite() != null && ctx.suite().statement() != null) {
        for (HtmlParser.StatementContext stmtCtx : ctx.suite().statement()) {
            body.add((StatementNode) visit(stmtCtx));
        }
    }
    
    return new FunctionDefNode(token.getLine(), token.getCharPositionInLine(),
            name, parameters, body, decorators);
}

5.2 
SymbolTableVisitor.java
الغرض: المرور على AST وملء جدول الرموز

مثال - زيارة تعريف دالة:

@Override
public Void visitFunctionDefNode(FunctionDefNode node) {
    // تعريف الدالة في النطاق الحالي
    symbolTable.define(node.getName(), "function", "function", 
                       node.getLine(), node.getColumn());
    
    // الدخول إلى نطاق الدالة
    symbolTable.enterScope(node.getName(), "function");
    
    // إضافة المعاملات
    for (ParameterNode param : node.getParameters()) {
        param.accept(this);
    }
    
    // معالجة جسم الدالة
    for (StatementNode stmt : node.getBody()) {
        stmt.accept(this);
    }
    
    // الخروج من نطاق الدالة
    symbolTable.exitScope();
    
    return null;
}
مثال - زيارة متغير:

@Override
public Void visitAssignmentNode(AssignmentNode node) {
    String dataType = inferType(node.getValue());
    symbolTable.define(node.getVariable(), "variable", dataType,
                       node.getLine(), node.getColumn());
    if (node.getValue() != null) {
        node.getValue().accept(this);
    }
    return null;
}

@Override
public String visitHtml ElementNode(HtmlElementNode node) {
    StringBuilder sb = new StringBuilder();
    sb.append(indent()).append("HTML Element: ").append(node.getTagName());
    
    if (node.isDoctype()) {
        sb.append(" (DOCTYPE)");
    }
    if (node.isSelfClosing()) {
        sb.append(" (self-closing)");
    }
    sb.append("\\n");
    
    // طباعة السمات
    if (!node.getAttributes().isEmpty()) {
        indentLevel++;
        for (AttributeNode attr : node.getAttributes()) {
            sb.append(attr.accept(this));
        }
        indentLevel--;
    }
    
    // طباعة العناصر الفرعية
    if (!node.getChildren().isEmpty()) {
        indentLevel++;
        for (ElementNode child : node.getChildren()) {
            sb.append(child.accept(this));
        }
        indentLevel--;
    }
    
    return sb.toString();
}

الوظائف:

طباعة النتائج
كود مبسط:

private static void processFile(String filename) {
    // 1. قراءة الملف
    String content = readFile( filename);
    
    // 2. إنشاء المحلل اللغوي والنحوي
    CharStream input = CharStreams.fromString(content);
    HtmlLexer lexer = new HtmlLexer(input);
    CommonTokenStream tokens = new CommonTokenStream(lexer);
    HtmlParser parser = new HtmlParser(tokens);
    
    // 3. تحليل المدخلات
    HtmlParser.StartContext tree = parser.start();
    
    // 4. بناء AST
    ASTBuilder astBuilder = new ASTBuilder();
    DocumentNode documentNode = astBuilder.visit(tree);
    
    // 5. طباعة AST
    PrintVisitor printVisitor = new PrintVisitor();
    String astOutput = documentNode.accept(printVisitor);
    System.out.println(astOutput);
    
    // 6. بناء جدول الرموز  
    SymbolTable symbolTable = new SymbolTable();
    SymbolTableVisitor symbolTableVisitor = new SymbolTableVisitor(symbolTable);
    documentNode.accept(symbolTableVisitor);
    
    // 7. طباعة جدول الرموز
    symbolTable.printSymbolTable();
}
